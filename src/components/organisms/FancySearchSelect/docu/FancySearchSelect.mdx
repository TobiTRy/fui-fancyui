### Overview

The FancySearchSelect component provides a sophisticated search and select interface with dropdown functionality. It combines a text input with a filterable list of items, supporting both keyboard and mouse navigation. This component is perfect for creating searchable dropdowns, auto-complete fields, and advanced selection interfaces.

### Key Features

- **Advanced Search Filtering**: Built-in text matching or custom filter functions
- **Keyboard Navigation**: Full arrow key, Tab, Enter, and Escape support
- **Smart Navigation Modes**: Automatic switching between mouse and keyboard navigation
- **Flexible Item Display**: Multiple layout modes for items with icons, images, and descriptions
- **Customizable Behavior**: Options for open-on-focus, clear-on-select, and more
- **Accessibility Ready**: Proper focus management and keyboard interaction patterns
- **Theme Integration**: Full design system integration with theme types and layers

### Initial Setup

1. **Import the Component**

```tsx
import { FancySearchSelect } from '@letsfollowme/fui-fancyui';
import type { TSearchSelectItem } from '@letsfollowme/fui-fancyui';
```

2. **Use the Component**

```tsx
const items: TSearchSelectItem[] = [
  {
    id: '1',
    title: 'Option 1',
    description: 'First option description',
  },
  {
    id: '2',
    title: 'Option 2',
    description: 'Second option description',
  },
];

<FancySearchSelect items={items} placeholder="Search options..." onSelect={(item) => console.log('Selected:', item)} />;
```

### Advanced Implementation Example

```tsx
import React, { useState } from 'react';
import { FancySearchSelect } from '@letsfollowme/fui-fancyui';
import type { TSearchSelectItem } from '@letsfollowme/fui-fancyui';

const AdvancedSearchSelect = () => {
  const [selectedValue, setSelectedValue] = useState('');

  const items: TSearchSelectItem[] = [
    {
      id: 'user1',
      title: 'John Doe',
      description: 'Senior Developer',
      icon: <UserIcon />,
    },
    {
      id: 'user2',
      title: 'Jane Smith',
      description: 'Product Manager',
      icon: <UserIcon />,
    },
  ];

  const customFilter = (items: TSearchSelectItem[], searchValue: string) => {
    return items.filter((item) => {
      const searchLower = searchValue.toLowerCase();
      return (
        item.title.toLowerCase().includes(searchLower) ||
        item.description?.toLowerCase().includes(searchLower) ||
        item.id.toString().includes(searchLower)
      );
    });
  };

  return (
    <FancySearchSelect
      items={items}
      value={selectedValue}
      onChange={(e) => setSelectedValue(e.target.value)}
      onSelect={(item) => {
        setSelectedValue(item.title);
        console.log('Selected user:', item);
      }}
      filterFunction={customFilter}
      placeholder="Search users..."
      noItemsText="No users found"
      maxHeight="250px"
      openOnFocus={true}
      itemLayoutMode="span"
      themeType="primary"
      layer={2}
    />
  );
};
```
